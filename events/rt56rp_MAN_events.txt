country_event = {
	id = MAN_rp_event.3
	title = MAN_rp_event.3.t
	desc = MAN_rp_event.3.desc
	
	is_triggered_only = yes 
	
	#ACCEPT
	option = {
		name = MAN_rp_event.3.a
		IF = {
			limit = { has_country_flag = MAN_economic_demands_tt }
			remove_ideas = MAN_communist_cells
		}
		custom_effect_tooltip = MAN_puyi_5_tt
		add_to_variable = { MAN_puyi_loyalists = 5 }
		multiply_variable = {MAN_puyi_loyalists = -0.01}
		clamp_variable = {
			var = MAN_puyi_loyalists
			min = -0.9
			max = -0.1
		}
		add_to_variable = { MAN_puyi_loyalists = 1}
		start_civil_war = {
			ideology = fascism
			ruling_party = neutrality
			size = var:MAN_puyi_loyalists
			states = all
			states_filter = {
				NOT = { has_state_flag = MAN_infiltrated }
			}
			keep_all_characters = yes
			load_focus_tree = {
				tree = manchukuo_focus_new
				keep_completed = yes
				copy_completed_from = ROOT
			}
		}
		MAN_doihara_kenji = {set_nationality = FMA}
		MAN_kishi_nobosuke  = {set_nationality = FMA }
		MAN_ruan_zhenduo = { set_nationality = FMA }

		FMA = { 
			complete_national_focus = MAN_depose_the_weak_monarch_2
			set_country_flag = MAN_puyi_loyal
			set_country_flag = MAN_puyi_couped
			add_stability = -0.3
			add_ideas = limited_conscription
			add_ideas = partial_economic_mobilisation
			add_ideas = limited_exports
			set_power_balance = {
				id = MAN_kwantung_bop
				left_side = MAN_powerful_kwantung
				right_side = MAN_dead_kwantung
			}
		}
		hidden_effect = {
			JAP = {
				declare_war_on = {
					target = ROOT
					type = annex_everything
				}
			}
			set_political_party = {
				ideology = fascism
				popularity = 10
			}
		}
		remove_country_leader_trait = puppet_emperor
		add_country_leader_trait = MAN_emperor_of_china
		complete_national_focus = MAN_the_qing_restoration
		IF = {
			limit = { has_country_flag = MAN_bandits_allied }
			division_template = {
				name = "Partisan Armies"
				priority = 1
				is_locked = no
				division_names_group = MAN_INF_01
				regiments = {
					irregular_infantry = { x = 0 y = 0 }
					irregular_infantry = { x = 0 y = 1 }
					irregular_infantry = { x = 0 y = 2 }

					irregular_infantry = { x = 1 y = 0 }
					irregular_infantry = { x = 1 y = 1 }
					irregular_infantry = { x = 1 y = 2 }
				}
			}
			set_country_leader_ideology = military_junta
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"1. Partisan Formation\" division_template = \"Partisan Armies\" start_experience_factor = 0.7" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"2. Partisan Formation\" division_template = \"Partisan Armies\" start_experience_factor = 0.7" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"3. Partisan Formation\" division_template = \"Partisan Armies\" start_experience_factor = 0.7" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"4. Partisan Formation\" division_template = \"Partisan Armies\" start_experience_factor = 0.7" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"5. Partisan Formation\" division_template = \"Partisan Armies\" start_experience_factor = 0.7" 
					owner = MAN
				}
			}
		}
		IF = {
			limit = { has_country_flag = MAN_puyi_ajnsa }
			division_template = {
				name = "Anti-Japanse National Salvation Army"
				priority = 1
				is_locked = no
				division_names_group = MAN_INF_01
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }

					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
					infantry = { x = 1 y = 2 }
					
					infantry = { x = 2 y = 0 }
					infantry = { x = 2 y = 1 }
					infantry = { x = 2 y = 2 }
					
					infantry = { x = 3 y = 0 }
					infantry = { x = 3 y = 1 }
					infantry = { x = 3 y = 2 }
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"1. National Salvation Brigade\" division_template = \"Anti-Japanse National Salvation Army\" start_experience_factor = 1" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"2. National Salvation Brigade\" division_template = \"Anti-Japanse National Salvation Army\" start_experience_factor = 0.9" 
					owner = MAN
				}
			}
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"3. National Salvation Brigade\" division_template = \"Anti-Japanse National Salvation Army\" start_experience_factor = 0.96" 
					owner = MAN
				}
			}
		}
		IF = {
			limit = { has_country_flag = MAN_joint_planning }
			add_timed_idea = {
				idea = MAN_the_hsinking_plan
				days = 100
			}
			
			IF = {
				limit = { CHI = { has_war_with = JAP } }
				CHI = {
					add_timed_idea = {
						idea = MAN_the_hsinking_plan
						days = 50
					}
				}
			}
		}
		IF = {
			limit = { has_country_flag = MAN_seperatism }
			CHI = {
				send_equipment = {
					equipment = infantry_equipment
					amount = 10000
					target = ROOT
				}
			}
		}
		IF = {
			limit = { has_country_flag = MAN_buff }
			JAP = {
				add_timed_idea = {
					idea = MAN_infiltrated_communications_network
					days = 75
				}
			}
		}
		IF = {
			limit = { has_country_flag = MAN_xueliang_sent}
			every_possible_country = {
				limit = {
					has_character = CHI_zhang_xueliang
				}
				random_character = {
					limit = { is_character = CHI_zhang_xueliang }
					set_nationality = MAN
				}
			}
			CHI = {
			transfer_units_fraction = {
				target = MAN
				size = 0
				army_ratio = 0.04
				keep_unit_leaders_trigger = { always = yes}
			}
			}
			add_advisor_role = {
				character = CHI_zhang_xueliang
				activate = yes
				advisor = {
					slot = army_chief
					cost = 50
					idea_token = MAN_zhang_xueliang
					traits = {
						MAN_leader_of_the_northeast_army
					}
				}
			}	
		}
	}
}

		name = MAN_rp_event.5.a

		MAN = {
			country_event = CHI_warlords.19

				target = ROOT

			}
		}
	}
	option = {
		name = MAN_rp_event.5.b
		every_state = {
			IF = {
				limit = {
					owner = {
						OR = {
							original_tag = ROOT
							is_subject_of = ROOT
						}
					}
					is_core_of = MAN
				}
				transfer_state_to = MAN
			}
		}
		MAN = {
			country_event = CHI_warlords.20
		}
	}

}
