country_event = {
	id = MAN_rp_event.3
	title = MAN_rp_event.3.t
	desc = MAN_rp_event.3.desc
	
	is_triggered_only = yes 
	
	#ACCEPT
	option = {
		name = MAN_rp_event.3.a
		custom_effect_tooltip = MAN_puyi_5_tt
		add_to_variable = { MAN_puyi_loyalists = 5 }
		multiply_variable = {MAN_puyi_loyalists = -0.01}
		clamp_variable = {
			var = MAN_puyi_loyalists
			min = -0.9
			max = -0.1
		}
		add_to_variable = { MAN_puyi_loyalists = 1}
		start_civil_war = {
			ideology = fascism
			ruling_party = neutrality
			size = var:MAN_puyi_loyalists
			states = all
			states_filter = {
				NOT = { has_state_flag = MAN_infiltrated }
			}
			keep_all_characters = yes
			set_country_flag = MAN_puyi_loyal
			set_country_flag = MAN_puyi_couped
			complete_national_focus = MAN_depose_the_weak_monarch_2
			add_stability = -0.3
			load_focus_tree = {
				tree = manchukuo_focus_new
				keep_completed = yes
				copy_completed_from = ROOT
			}
		}
		hidden_effect = {
			set_political_party = {
				ideology = fascism
				popularity = 10
			}
		}
		remove_country_leader_trait = puppet_emperor
		add_country_leader_trait = MAN_emperor_of_china
		complete_national_focus = MAN_the_qing_restoration

	}
}

			}
		}
		MAN = {

		}
