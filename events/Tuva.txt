country_event = { # Choice to send peace offer
	id = tuva.9
	title = tuva.9.t
	desc = tuva.9.d
	picture = GFX_report_event_generic_communist_militia
	fire_only_once = yes
    trigger = {
		has_country_flag = TAN_altai_war_mechanic
		any_enemy_country = {
			NOT = {
				tag = ALT
			}
			owns_state = 654
		}
		controls_state = 654
    }
	
	option = { # Offer peace
		name = tuva.9.a
			every_enemy_country = {
				limit = {
					owns_state = 654
				}
				country_event = {
					id = tuva.10
					days = 1
				}
			}
	}
	option = { # Continue war
		name = tuva.9.b
		ai_chance = { base = 0 }
	}
}
