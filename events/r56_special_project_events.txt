add_namespace = r56_special_project

# Special project events (currently just about helicopters)
# Event 1, triggers upon completion of the Gotterdammerung Helicopter Special Project. Otherwise same reward done through the decision.
country_event = {
	id = r56_special_project.1
	title = r56_special_project.1.t
	desc = r56_special_project.1.desc
	picture = GFX_report_event_generic_research_lab
	
	is_triggered_only = yes
	
	option = { # Fund the vertical flight company
		name = r56_special_project.1.a
		trigger = {
			has_political_power > 99
			has_air_experience > 49
		}
		ai_chance = {
			base = 0
		}
        
        air_experience = -50
		add_political_power = -100

		custom_effect_tooltip = r56_continue_the_study_of_vertical_flight_tt
		if = {
			limit = {
				has_dlc = "Arms Against Tyranny"
			}
			custom_effect_tooltip = r56_continue_the_study_of_vertical_flight_AAT_tt
		}		
		
		hidden_effect = {
			set_country_flag = r56_generic_helicopter_company_flag
			if = {
				limit = {
					has_dlc = "Arms Against Tyranny"
					ROOT = {
						any_scientist = { # check if it works
							has_trait = scientist_trait_helicopter_expert
						}
					}
				}
				#r56_generic_helicopter_company -> no need as generic one is there
				ROOT = {
					owner = {
						mio:generic_helicopter_organization = { # base MIO copied by countries with unique companies
							add_mio_size = 1
						} 				
					}
				}
			}

			if = {
				limit = {
					has_dlc = "Arms Against Tyranny"
					OR = {
						has_idea_with_trait = war_industrialist
						has_idea_with_trait = minister_of_aircraft_production
					}   
				}
				ROOT = {
					owner = {
						mio:generic_helicopter_organization = { # base MIO copied by countries with unique companies
							add_mio_funds = 1000
						}
					}
				}
			}			
		}

		country_event = { id = r56_special_project.2 }
	}
	option = { # Refuse
		name = r56_special_project.1.b
		ai_chance = {
			base = 1
		}
        # effect_tooltip show decision
		unlock_decision_tooltip = r56_continue_the_study_of_vertical_flight
	}
}
