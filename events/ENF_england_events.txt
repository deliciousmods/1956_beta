# England - Demand Ireland
country_event = {
	id = englandmod.5
	title = englandmod.5.t
	desc = englandmod.5.d
	picture = GFX_report_event_king_speech

	is_triggered_only = yes

	option = {
		name = englandmod.5.a
		ai_chance = {
			base = 80
			modifier = {
				factor = 0
				any_other_country = {
					is_major = yes
					NOT = { tag = ENG }
					OR = {
						has_guaranteed = IRE
						is_in_faction_with = IRE
					}
				}
			}
		}
		custom_effect_tooltip = GAME_OVER_TT
		ENG = {
			effect_tooltip = {
				annex_country = {
					target = IRE
					transfer_troops = no
				}
			}
			country_event = englandmod.6
		}
	}

	option = {
		name = englandmod.5.b
		ai_chance = {
			base = 20
			modifier = {
				factor = 0
				OR = {
					AND = {
						is_in_faction_with = ENG
						ENG = { is_faction_leader = yes }
					}
					AND = {
						is_in_faction = no 
						NOT = {
							any_other_country = {
								NOT = { tag = ENG }
								has_guaranteed = IRE
							} 
						}
					}
				}
			}
		}
		ENG = {
			effect_tooltip = {
				create_wargoal = {
					type = annex_everything
					target = IRE
				}
			}
			country_event = englandmod.7
		}
	}
}

# England - Invite Japan To Join Faction
country_event = {
	id = englandmod.17
	title = englandmod.17.t
	desc = englandmod.17.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	
	option = {
		name = englandmod.17.a
		ai_chance = { base = 95 }
		ENG = {
			if = {
				limit = { is_in_faction = no }
				create_faction_from_template = {
					template = faction_template_generic_dominance
					name = imperial_fascist_league_faction
					icon = GFX_faction_logo_generic_fascist
				}
			}
		}
		ENG = {	add_to_faction = JAP }
		JAP = {
			add_ai_strategy = {
				type = alliance
				id = "ENG"
				value = 200
			}
		}
		ENG = {	
			country_event = { id = englandmod.18 days = 1 }
		}
	}
	option = {
		name = englandmod.17.b
		ai_chance = { base = 5 }
		ENG = {	
			country_event = { id = englandmod.19 days = 1 }
		}
	}
}

	}
