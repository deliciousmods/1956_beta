country_event = {
	id = GXC_event.1
	title = GXC_event.1.t
	desc = GXC_event.1.desc
	picture = GFX_report_event_merchant_ship_01
	
	is_triggered_only = yes

	
	#ACCEPT
	option = {
		name = GXC_event.1.a
		set_capital = {
			state = 594
			remember_old_capital = no
		}
		every_controlled_state = {
			limit = {
				NOT = {
					state = 599
					state = 594
					AND = {
						state = 1112
						controller = { has_country_flag = GXC_nanlu_gxc }
					}
				}
			}
			transfer_state_to = CHI
			remove_core_of = GXC
		}
		IF = {
			limit = {
				has_idea = GXC_aftershock_of_hu_hanmins_death
			}
			remove_ideas = GXC_aftershock_of_hu_hanmins_death
		}
		else_if = {
			limit = { 
				has_idea = GXC_hu_hanmins_legacy
			}
			remove_ideas = GXC_hu_hanmins_legacy
		}
		drop_cosmetic_tag = yes
		retire_character = GXC_chen_jitang
		complete_national_focus = GXC_aftermath_of_the_liangguang_incident
	}
}
