# Change here is that Quezon ages abit
country_event = {
	id = phil.12
	title = SEA_PHI_political_events.63.t
	desc = SEA_PHI_political_events.63.desc
	picture = GFX_report_event_election_vote

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			country_event = {	#Fired earlier when liberated from exile
				id = SEA_PHI_political_events.64 
				days = 2190
			}
		}
		if = {
			limit = { has_character = PHI_manuel_luis_quezon }
			set_portraits = {
				character = PHI_manuel_luis_quezon
				civilian = { large ="GFX_portrait_PHI_manuel_quezon" }
			}
		}
	}

	option = {	#Keep leader
		name = SEA_PHI_political_events.63.keep_leader
		ai_chance = {
			base = 1
		}
		add_political_power = 25
		hidden_effect = {
			if = {
				limit = {
					has_country_leader = { 
						character = PHI_manuel_luis_quezon
						ruling_only = yes
					}
				}
				country_event = {	#No Quezon death event so fire the election early instead
					id = SEA_PHI_political_events.64
					days = 1000
				}
			}
		}
	}

	option = {
		name = SEA_PHI_political_events.63.roxas
		trigger = {
			has_character = PHI_manuel_roxas
			PHI_manuel_roxas = {
				is_country_leader = no
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
		PHI_manuel_roxas = {
			if = {
				limit = {
					can_be_country_leader = yes
				}
				promote_character = yes
			}
			else = {
				add_country_leader_role = {
					promote_leader = yes
					country_leader = {
						ideology = conservatism
						expire = "1965.1.1.1"
						traits = { PHI_political_hacendado }
					}
				}
			}
		}
	}

	option = {	#Laurel
		name = SEA_PHI_political_events.63.laurel
		trigger = {
			has_character = PHI_jose_laurel
			PHI_jose_laurel = {
				is_country_leader = no
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
		PHI_jose_laurel = {
			if = {
				limit = {
					can_be_country_leader = yes
				}
				promote_character = yes
			}
			else = {
				add_country_leader_role = {
					promote_leader = yes
					country_leader = {
						ideology = conservatism
						expire = "1965.1.1.1"
						traits = { PHI_social_reformer }
					}
				}
			}
		}
	}
}
