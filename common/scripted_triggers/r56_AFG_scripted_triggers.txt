r56_AFG_check_country_relationship = { # This is used for diplo, checks relationship between countries that AFG interacts with
	OR = {
		tag = SOV
		tag = ENG
		tag = GER
		tag = ITA
		AND = {
			OR = {
				tag = JAP
				AND = {
					tag = USA
					OR = {
						has_war_with = FROM
						# TODO: Add check for AFG-USA relations
					}
				}
			}
			has_country_flag = r56_AFG_established_diplomatic_relations
		}
	}
	NOT = {
		has_capitulated = yes
		has_war_with = ROOT
		is_ally_with = ROOT
	}
	FROM = { # Nations are not allied
		NOT = {
			is_ally_with = PREV
		}
	}
	OR = {
		AND = { # PREV is Anti-Communist
			has_government = neutrality
			FROM = {
				has_government = communism
			}
		}
		AND = { # PREV is Anti-Extremist
			has_government = democratic
			FROM = {
				OR = {
					has_government = communism
					has_government = fascism
				}
			}
		}
		AND = { # PREV is Anti-Bolshevik and Anti-Democratic
			has_government = fascism
			FROM = {
				OR = {
					has_government = democratic
					has_government = communism
				}
			}
		}
		AND = { # PREV is Anti-Everything
			has_government = communism
			FROM = {
				NOT = {
					has_government = communism
				}
			}
		}
	}
}
