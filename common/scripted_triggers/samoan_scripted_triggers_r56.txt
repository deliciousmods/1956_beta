#scripted triggers for Samoa

SAM_is_formal_subject_of_nzl = {
	original_tag = SAM
	OR = {
		has_cosmetic_tag = SAM_TWS
		is_subject_of = NZL
		is_subject_of = ENG
	}
	NOT = { has_cosmetic_tag = SAM_MAU }
}

SAM_is_formally_independent = {
	original_tag = SAM
	OR = {
		is_subject = no
		has_cosmetic_tag = SAM_MAU
	}
}
