# Army
CHI_regionalists_winning_power_struggle = {
	custom_trigger_tooltip = {
		tooltip = KMT_power_struggle_easier_tt
	}
	CHI = {
		IF = {
			limit = {
				has_country_flag = {
					flag = KMT_power_struggle_flag
					days > 150
				}
			}
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.1
			}
		}
		else_if = {
			limit = {
				has_country_flag = {
					flag = KMT_power_struggle_flag
					days > 120
				}
			}
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.25
			}
		}
		else_if = {
			limit = {
				has_country_flag = {
					flag = KMT_power_struggle_flag
					days > 90
				}
			}
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.4
			}
		}
		else_if = {
			limit = {
				has_country_flag = {
					flag = KMT_power_struggle_flag
					days > 60
				}
			}
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.55
			}
		}
		else_if = {
			limit = {
				has_country_flag = {
					flag = KMT_power_struggle_flag
					days > 30
				}
			}
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.6
			}
		}
		else = {
			power_balance_value = {
				id = CHI_cks_cris_bop
				value < -0.75
			}
		}
	}
}
