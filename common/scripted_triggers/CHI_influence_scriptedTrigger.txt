# Army

CHI_army_faction_is_tier_1 = {
	custom_trigger_tooltip = {
		tooltip = CHI_army_faction_is_tier_1_tt
		OR = {
			has_idea = CHI_army_faction_tier_1
			has_idea = CHI_army2_faction_tier_1
			AND = {
				has_completed_focus = KMT_a_strong_nation_with_strong_regions
				check_variable = {
					CHI.KMT_yun_influence_modifier = 0.025
				}
			}
		}
