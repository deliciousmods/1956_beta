BEL_historical_ai = {
	OR = {
		AND = {
			is_historical_focus_on = yes
			has_game_rule = {
				rule = BEL_AI_BEHAVIOR
				option = DEFAULT
			}
		}
		has_country_flag = BEL_AI_RANDOM_DEMOCRATIC_HISTORICAL
		has_game_rule = {
			rule = BEL_AI_BEHAVIOR
			option = DEMOCRATIC_HISTORICAL
		}
	}
}
