is_monarchist = {
	OR = {
		AND = {
			original_tag = ENG
			OR = {
				has_idea = george_v
				has_idea = ENG_king_edward
				has_idea = ENG_king_edward_fascism
				has_idea = ENG_george_vi
				has_completed_focus = ENG_the_kings_party
			}
		}
		AND = {
			original_tag = GER
			has_completed_focus = GER_revive_the_kaiserreich
		}
		AND = {
			original_tag = JAP
			OR = {
				has_idea = JAP_hirohito
				has_country_leader = {
					character = JAP_emperor_hirohito
					ruling_only = yes
				}
				has_country_leader = {
					character = JAP_empress_nagako
					ruling_only = yes
				}
				has_country_leader = {
					character = JAP_emperor_yasuhito
					ruling_only = yes
				}
				has_country_leader = {
					character = JAP_emperor_nobuhito
					ruling_only = yes
				}
				has_completed_focus = JAP_revere_the_emperor_destroy_the_traitors
			}
		}
		AND = {
			original_tag = SOV
			has_completed_focus = SOV_beaten_but_not_defeated
		}
		AND = {
			original_tag = FRA
			has_completed_focus = FRA_the_council_of_rambouillet
		}
		AND = {
			original_tag = ITA
			OR = {
				has_idea = vittorio_emanuele
				has_idea = vittorio_emanuele_emperor_of_ethiopia
				has_idea = vittorio_emanuele_king_of_albania
				has_idea = vittorio_emanuele_albania_ethiopia
				has_country_leader = {
					character = ITA_vittorio_emanuele_iii
					ruling_only = yes
				}
				has_country_leader = {
					character = ITA_prince_umberto
					ruling_only = yes
				}
			}
		}
		AND = {
			original_tag = GRE
			OR = {
				AND = {
					NOT = { has_dlc = "Battle for the Bosporus" }
					has_cosmetic_tag = GRE
				}
			}
		}
		has_country_leader_ideology = emperor_fascism
		has_country_leader_ideology = absolute_monarchism
		has_country_leader_ideology = constitutional_monarchy
		}
	}
}

INS_is_enough_built_east = {
	infrastructure > 2
	industrial_complex > 0
	OR = {
		has_dynamic_modifier = { modifier = advanced_develop_modif_middle }
		has_dynamic_modifier = { modifier = advanced_develop_modif_east }
	}
}

INS_is_enough_built_west = {
	infrastructure > 3
	OR = {
		industrial_complex > 1
		arms_factory > 1
		AND = {
			industrial_complex > 0
			arms_factory > 0
		}
	}
	OR = {
		has_dynamic_modifier = { modifier = advanced_develop_modif_middle }
		has_dynamic_modifier = { modifier = advanced_develop_modif_west }
	}
}

INS_political_grouping_allowed = {
	OR = {
		if = {
			limit = {
				has_character_flag = INS_stuwgroep_member_flag
			}
			INS = {
				has_completed_focus = INS_r56_reinstate_stuwgroep
			}
		}
		else_if = {
			limit = {
				has_character_flag = INS_parindra_member_flag
			}
			INS = {
				OR = {
					has_completed_focus = INS_r56_placate_nationalists
					has_completed_focus = INS_r56_appease_nationalists
				}
			}
		}
		else_if = {
			limit = {
				has_character_flag = INS_psii_member_flag
			}
			INS = {
				has_completed_focus = INS_r56_invite_psii
			}
		}
		else_if = {
			limit = {
				has_character_flag = INS_nip_member_flag
			}
			INS = {
				has_completed_focus = INS_r56_indies_nation
			}
		}
		else_if = {
			limit = {
				has_character_flag = INS_gerindo_member_flag
			}
			INS = {
				has_country_flag = INS_gerindo_allowed_flag
			}
		}
		else_if = {
			limit = {
				has_character_flag = INS_ppbb_member_flag
			}
			INS = {
				OR = {
					has_completed_focus = INS_r56_support_sutarjo
					has_completed_focus = INS_r56_reform_bb
				}
			}
		}
	}
}
