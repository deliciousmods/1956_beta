#makes facists and commie countries loath ancap HON
HON_exterminate_ancaps = {
	allowed = {
		exists = yes
	}

	enable = {
		country_exists = HON
		OR = {
			has_government = fascism
			has_government = communism
		}
		HON = {
			has_completed_focus = HON_free_our_markets
			exists = yes
		}
		NOT = {
			original_tag = HON
		}
	}

	abort = {
		OR = {
			is_subject_of = HON
			has_war_with = HON
		}
	}

	ai_strategy = {
		type = antagonize
		id = HON
		value = 125

#Befriend soviets if go communist
HON_simp_for_soviets = {
	allowed = {
		original_tag = HON
	}

	enable = {
		ROOT = {
			exists = yes
			has_completed_focus = HON_go_communist
		}
		SOV = {
			has_government = communism
		}
	}
	
	abort = {
		SOV = { has_war_with = HON }

		type = befriend

	ai_strategy = {
		type = support
		id = SOV
		value = 50
	}
	ai_strategy = {
		type = alliance
		id = SOV
		value = 25
