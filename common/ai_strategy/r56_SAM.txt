SAM_screw_new_zealand = {
	allowed = { original_tag = SAM }

	enable = {
		SAM = { exists = yes }
		NZL = { NOT = { has_completed_focus = NZL_the_goodwill_mission } }
	}
	
	abort = { NZL = { has_completed_focus = NZL_the_goodwill_mission } }
	
	ai_strategy = {
		type = befriend
		id = NZL
		value = -999
	}
}
