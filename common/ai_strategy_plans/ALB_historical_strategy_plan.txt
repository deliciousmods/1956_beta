ALB_historical_plan = {
	name = "ALB: Albanian Historical AI Plan"
	desc = "Historical monarchist behavior for Albania."

	allowed = {
		original_tag = ALB
	}

	enable = {
		OR = {
			AND = {
				is_historical_focus_on = yes
				has_game_rule = {
					rule = ALB_ai_behavior
					option = DEFAULT
				}
			}
			has_country_flag = ALB_AI_RANDOM_MONARCHIST_HISTORICAL
			has_game_rule = {
				rule = ALB_ai_behavior
				option = MONARCHIST_HISTORICAL
			}
		}
	}

	#Makes the AI completely cancel the set behavior.
	abort = {
	}

	ai_national_focuses = {
		ALB_political_reform
		ALB_vive_le_roi
		ALB_constitution_rule
		ALB_righteous_rule_focus
		ALB_royal_institute
		ALB_strategic_diplomacy
		ALB_trade
		ALB_ita_pre
		ALB_ita_affair
		#order below previously wrong and with dupe
		ALB_ita_strict
		ALB_will_to_fight
		ALB_form_generals
		ALB_no_ita_officer
		ALB_fortify_coastline
		ALB_yug_borders
	}

	focus_factors = {
		ALB_the_king_shall_be_desposed = 0
		ALB_absolute_rule = 0
		ALB_join_axis = 0
		ALB_defense_pact = 0
		ALB_strike_first = 0
	}

	#Keep small, as it is used as a factor for some things, such as research needs.
	#Recommended around 1.0. Useful for relation between plans.
	weight = {
		factor = 1.0
		modifier = {
			factor = 1.0
		}
	}
}
