FIN_r56_communist_plan = {
	name = "FIN: Finnish Communist AI Plan"
	desc = "Communist behavior for Finland."

	allowed = {
		original_tag = FIN
	}

	enable = {
		NOT = {
			has_dlc = "Arms Against Tyranny"
		}
		OR = {
			has_country_flag = ARG_AI_RANDOM_COMMUNIST
			has_game_rule = {
				rule = FIN_ai_behavior
				option = COMMUNIST
			}
		}
	}

	#Makes the AI completely cancel the set behavior.
	abort = {
		is_puppet = yes
		NOT = {
			has_government = communism
		}
	}

	ai_national_focuses = {
		fin_comfoc
	}

	focus_factors = {
		FIN_finnish_neutrality = 0
		FIN_right_wing_policies = 0
		fin_nationalismfoc = 0
	}

	#Keep small, as it is used as a factor for some things, such as research needs.
	#Recommended around 1.0. Useful for relation between plans.
	weight = {
		factor = 1.0
		modifier = {
			factor = 1.0
		}
	}
}
