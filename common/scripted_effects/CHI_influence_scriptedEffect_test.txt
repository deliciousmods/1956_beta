                        limit = {
                            YUN = {
                                exists = yes
                            }
                    }

                            every_character = {
                                set_nationality = CHI
                            }
                        }
                        annex_country = {
                            target = YUN
                            transfer_troops = yes
                        }
                        every_state = {
                            limit = {
                                is_core_of = YUN
                            }
                            remove_core_of = YUN
                        }
                        change_tag_from = YUN
                    }
                }
            }
        }
