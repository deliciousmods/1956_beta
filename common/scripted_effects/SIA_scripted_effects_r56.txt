SIA_improve_illiteracy_r56 = {
	if = {
		limit = { has_idea = SIA_high_illiteracy_r56 }
		swap_ideas = {
			remove_idea = SIA_high_illiteracy_r56
			add_idea = SIA_high_illiteracy_1_r56
		}
	}
	else_if = {
		limit = { has_idea = SIA_high_illiteracy_1_r56 }
		swap_ideas = {
			remove_idea = SIA_high_illiteracy_1_r56
			add_idea = SIA_illiteracy_r56
		}
	}
	else_if = {
		limit = { has_idea = SIA_high_illiteracy_1_r56 }
		remove_ideas = SIA_illiteracy_r56
		add_timed_idea = {
			days = 720
			idea = SIA_the_fight_against_illiteracy_r56
		}
	}
}
