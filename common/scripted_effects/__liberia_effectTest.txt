CHI_army_faction_gain = {
	if = {
	    limit = {
		    CHI_army_faction_is_tier_1 = yes
		}
		if = {
		    limit = { has_idea = CHI_army_faction_tier_1 }
		    swap_ideas = {
                remove_idea = CHI_army_faction_tier_1
                add_idea = CHI_army_faction_tier_2
            }
        }
        else_if = {
		    limit = { has_idea = CHI_army2_faction_tier_1 }
		    swap_ideas = {
                remove_idea = CHI_army2_faction_tier_1
                add_idea = CHI_army2_faction_tier_2
            }
        }
	}
	else_if = {
	    limit = {
		    CHI_army_faction_is_tier_2 = yes
		}
		swap_ideas = {
            remove_idea = CHI_army_faction_tier_2
            add_idea = CHI_army_faction_tier_3
        }
	}
	else_if = {
	    limit = {
		    CHI_army_faction_is_tier_3 = yes
		}
		swap_ideas = {
            remove_idea = CHI_army_faction_tier_3
            add_idea = CHI_army_faction_tier_4
        }
	}
}

		}
		swap_ideas = {
            remove_idea = CHI_army_faction_tier_4

        }
	}
}
