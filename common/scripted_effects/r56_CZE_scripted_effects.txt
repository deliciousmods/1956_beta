CZE_increase_internal_woes = {
	if = {
		limit = {
			has_country_flag = CZE_internal_woes_lvl_1
		}
		custom_effect_tooltip = CZE_minorities_radicalisation_1_2_iw_tt
		hidden_effect = {
			clr_country_flag = CZE_internal_woes_lvl_1
			set_country_flag = CZE_internal_woes_lvl_2
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_2
			}
			custom_effect_tooltip = CZE_minorities_radicalisation_2_3_iw_tt
			hidden_effect = {
				clr_country_flag = CZE_internal_woes_lvl_2
				set_country_flag = CZE_internal_woes_lvl_3
			}
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_3
			}
			custom_effect_tooltip = CZE_minorities_radicalisation_3_4_iw_tt
			hidden_effect = {
				clr_country_flag = CZE_internal_woes_lvl_3
				set_country_flag = CZE_internal_woes_lvl_4
			}
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_4
			}
			custom_effect_tooltip = CZE_minorities_radicalisation_4_5_iw_tt
			hidden_effect = {
				clr_country_flag = CZE_internal_woes_lvl_4
				set_country_flag = CZE_internal_woes_lvl_5
			}
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_5
			}
			custom_effect_tooltip = CZE_minorities_radicalisation_5_6_iw_tt
			hidden_effect = {
				clr_country_flag = CZE_internal_woes_lvl_5
				set_country_flag = CZE_internal_woes_lvl_6
			}
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_6
			}
			custom_effect_tooltip = CZE_minorities_radicalisation_6_7_iw_tt
			hidden_effect = {
				clr_country_flag = CZE_internal_woes_lvl_6
				set_country_flag = CZE_internal_woes_lvl_7
			}
		}
		else_if = {
			limit = {
				has_country_flag = CZE_internal_woes_lvl_7
			}
		}
		else = {
			set_country_flag = CZE_internal_woes_lvl_1
		}
	}
}

CZE_remove_internal_woes = {
	custom_effect_tooltip = cz_tt_29
	hidden_effect = {
		clr_country_flag = CZE_internal_woes_lvl_1
		clr_country_flag = CZE_internal_woes_lvl_2
		clr_country_flag = CZE_internal_woes_lvl_3
		clr_country_flag = CZE_internal_woes_lvl_4
		clr_country_flag = CZE_internal_woes_lvl_5
		clr_country_flag = CZE_internal_woes_lvl_6
		clr_country_flag = CZE_internal_woes_lvl_7
		remove_dynamic_modifier = {
			modifier = CZE_internal_woes_dynamic_modifier
		}
	}
}
