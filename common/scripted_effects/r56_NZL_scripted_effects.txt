NZL_worker_unions_angrier = {
	if = {
		limit = {
			has_idea = NZL_angered_worker_unions_idea
		}
		remove_ideas = NZL_angered_worker_unions_idea
		add_timed_idea = {
			idea = NZL_militant_worker_unions_idea
			days = 1095
		}
		activate_mission = NZL_militant_worker_unions
	}
	else_if = {
		limit = {
			has_idea = NZL_discontent_worker_unions_idea
		}
		swap_ideas = {
			remove_idea = NZL_discontent_worker_unions_idea
			add_idea = NZL_angered_worker_unions_idea
		}
	}
	else_if = {
		limit = {
			NOT = { has_idea = NZL_discontent_worker_unions_idea }
			NOT = { has_idea = NZL_angered_worker_unions_idea }
			NOT = { has_idea = NZL_militant_worker_unions_idea }
		}
		add_ideas = NZL_discontent_worker_unions_idea
	}
}
