r56_modifier_ui_iterate = {
	if = {
		limit = {
			check_variable = {
				r56_modifier_x_pos < 500
			}
		}
		add_to_variable = {
			var = r56_modifier_x_pos
			value = 46
		}
	}
	else = {
		set_variable = {
			var = r56_modifier_x_pos
			value = @X_leftmost
		}
		add_to_variable = {
			var = r56_modifier_y_pos
			value = 44
		}
	}
}
