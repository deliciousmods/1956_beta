# Political programme stages replace each other; none of these bonuses stack.
BOL_refresh_falange_programme = {
 if = {
  limit = { has_government = fascism }
  hidden_effect = { remove_ideas = { BOL_fsb_corporatism BOL_falangist_state BOL_ayllu_order BOL_lost_sea_campaign BOL_military_socialism BOL_busch_labor_code BOL_worker_militias BOL_technocratic_junta BOL_revolutionary_export BOL_aymara_support BOL_a_new_national_pride BOL_social_constitution BOL_social_labour_charter } }


  if = {
   limit = { has_country_flag = BOL_falange_andean_choice }
   if = {
    limit = { OR = { has_country_flag = BOL_falange_continental_programme has_completed_focus = BOL_falange_continental_directorate } }
    BOL_apply_falange_andean_federation = yes
   }
   else_if = {
    limit = { has_country_flag = BOL_falange_professionalization_complete }
    BOL_apply_falange_andean_army = yes
   }
   else = { BOL_apply_falange_andean_columns = yes }
   if = { limit = { NOT = { has_cosmetic_tag = true_plurinational_bolivia } } set_cosmetic_tag = true_plurinational_bolivia }
   hidden_effect = { if = { limit = { has_country_leader_with_trait = BOL_professional_army_advocate } remove_country_leader_trait = BOL_professional_army_advocate } }
   if = { limit = { NOT = { has_country_leader_with_trait = BOL_andean_mobilizer } } add_country_leader_trait = BOL_andean_mobilizer }
  }
  else_if = {
   limit = { has_country_flag = BOL_falange_hispanist_choice }
   if = {
    limit = { OR = { has_country_flag = BOL_falange_continental_programme has_completed_focus = BOL_falange_continental_directorate } }
    BOL_apply_falange_expeditionary_army = yes
   }
   else_if = {
    limit = { has_country_flag = BOL_falange_professionalization_complete }
    BOL_apply_falange_professional_army = yes
   }
   else = { BOL_apply_falange_hispanist_cadres = yes }
   hidden_effect = { if = { limit = { has_country_leader_with_trait = BOL_andean_mobilizer } remove_country_leader_trait = BOL_andean_mobilizer } }
   if = { limit = { NOT = { has_country_leader_with_trait = BOL_professional_army_advocate } } add_country_leader_trait = BOL_professional_army_advocate }
  }
  else = { BOL_apply_falange_assault_cadres = yes }
  set_country_flag = BOL_falange_programme_v4
 }
}

BOL_claim_historic_frontiers = {
 every_state = {
  limit = { BOL_is_falangist_historic_region = yes NOT = { is_core_of = ROOT } }
  add_claim_by = ROOT
 }
}

BOL_refresh_workers_programme = {
 if = {
  limit = { has_government = communism has_country_flag = BOL_rosca_workers_route }
  hidden_effect = { if = { limit = { has_country_leader_with_trait = BOL_professional_army_advocate } remove_country_leader_trait = BOL_professional_army_advocate } }
  hidden_effect = { if = { limit = { has_country_leader_with_trait = BOL_andean_mobilizer } remove_country_leader_trait = BOL_andean_mobilizer } }
  hidden_effect = { remove_ideas = { BOL_military_socialism BOL_busch_labor_code BOL_worker_militias BOL_ayllu_order BOL_plurinational_democracy BOL_technocratic_junta BOL_revolutionary_export BOL_workers_supply_councils BOL_social_constitution BOL_social_labour_charter } }
  if = {
   limit = { OR = { has_country_flag = BOL_workers_international_programme has_completed_focus = BOL_congress_of_american_workers } }
   BOL_apply_workers_international_state = yes
  }
  else_if = {
   limit = { has_country_flag = BOL_workers_army_organized }
   BOL_apply_workers_revolutionary_state = yes
  }
  else_if = {
   limit = { has_country_flag = BOL_workers_constitution_enacted }
   BOL_apply_workers_planned_republic = yes
  }
  else = { BOL_apply_workers_transition = yes }
  set_country_flag = BOL_workers_programme_v4
 }
}

BOL_workers_campaign_PRU = {
 if = {
  limit = {
   country_exists = PRU is_subject = no
   NOT = { has_war_with = PRU }
   NOT = { is_in_faction_with = PRU }
   PRU = { NOT = { has_government = communism } NOT = { is_subject_of = ROOT } }
  }
  create_wargoal = { type = puppet_wargoal_focus target = PRU }
 }

 else_if = {
  limit = { country_exists = PRU PRU = { has_government = communism } NOT = { has_war_with = PRU } }
  add_opinion_modifier = { target = PRU modifier = BOL_diplomatic_friendship }
 }
}
