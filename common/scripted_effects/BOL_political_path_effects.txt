# Political programme stages replace each other; none of these bonuses stack.
BOL_refresh_falange_programme = {
 if = {
  limit = { has_government = fascism }
  remove_ideas = { BOL_workers_transition BOL_workers_planned_republic BOL_workers_revolutionary_state BOL_ayllu_mass_levies BOL_ayllu_peoples_army BOL_ayllu_road_army BOL_ayllu_confederate_levies BOL_falange_assault_cadres BOL_falange_hispanist_cadres BOL_falange_professional_army BOL_falange_andean_columns BOL_falange_andean_army BOL_fsb_corporatism BOL_falangist_state BOL_ayllu_order BOL_lost_sea_campaign BOL_military_socialism BOL_busch_labor_code BOL_state_mining_control BOL_worker_militias BOL_technocratic_junta BOL_revolutionary_export BOL_aymara_support BOL_a_new_national_pride }
  remove_country_leader_trait = BOL_professional_army_advocate
  remove_country_leader_trait = BOL_andean_mobilizer
  if = {
   limit = { has_country_flag = BOL_falange_andean_choice }
   if = {
    limit = { has_country_flag = BOL_falange_professionalization_complete }
    add_ideas = BOL_falange_andean_army
   }
   else = { add_ideas = BOL_falange_andean_columns }
   set_cosmetic_tag = true_plurinational_bolivia
   add_country_leader_trait = BOL_andean_mobilizer
  }
  else_if = {
   limit = { has_country_flag = BOL_falange_hispanist_choice }
   if = {
    limit = { has_country_flag = BOL_falange_professionalization_complete }
    add_ideas = BOL_falange_professional_army
   }
   else = { add_ideas = BOL_falange_hispanist_cadres }
   add_country_leader_trait = BOL_professional_army_advocate
  }
  else = { add_ideas = BOL_falange_assault_cadres }
  set_country_flag = BOL_falange_programme_v4
 }
}

BOL_claim_historic_frontiers = {
 every_state = {
  limit = { BOL_is_falangist_historic_region = yes NOT = { is_core_of = ROOT } }
  add_claim_by = ROOT
 }
}

BOL_refresh_workers_programme = {
 if = {
  limit = { has_government = communism has_country_flag = BOL_rosca_workers_route }
  remove_country_leader_trait = BOL_professional_army_advocate
  remove_country_leader_trait = BOL_andean_mobilizer
  remove_ideas = { BOL_falange_assault_cadres BOL_falange_hispanist_cadres BOL_falange_professional_army BOL_falange_andean_columns BOL_falange_andean_army BOL_ayllu_mass_levies BOL_ayllu_peoples_army BOL_ayllu_road_army BOL_ayllu_confederate_levies BOL_workers_transition BOL_workers_planned_republic BOL_workers_revolutionary_state BOL_military_socialism BOL_busch_labor_code BOL_state_mining_control BOL_worker_militias BOL_ayllu_order BOL_plurinational_democracy BOL_technocratic_junta BOL_revolutionary_export BOL_workers_supply_councils }
  if = {
   limit = { has_country_flag = BOL_workers_army_organized }
   add_ideas = BOL_workers_revolutionary_state
  }
  else_if = {
   limit = { has_country_flag = BOL_workers_constitution_enacted }
   add_ideas = BOL_workers_planned_republic
  }
  else = { add_ideas = BOL_workers_transition }
  set_country_flag = BOL_workers_programme_v4
 }
}

BOL_workers_campaign_PRU = {
 if = {
  limit = {
   country_exists = PRU is_subject = no
   NOT = { has_war_with = PRU }
   NOT = { is_in_faction_with = PRU }
   PRU = { NOT = { has_government = communism } NOT = { is_subject_of = ROOT } }
  }
  create_wargoal = { type = puppet_wargoal_focus target = PRU }
 }

 else_if = {
  limit = { country_exists = PRU PRU = { has_government = communism } NOT = { has_war_with = PRU } }
  add_opinion_modifier = { target = PRU modifier = BOL_diplomatic_friendship }
 }
}
