MAN_low_legitimacy_level_down2 = {
	if = {
		limit = { has_idea = MAN_low_legitimacy_12 }
		#Makes the tooltip clearer if player looks at the effect early in the game
		effect_tooltip = {
			swap_ideas = {
				remove_idea = MAN_low_legitimacy_22
				add_idea = MAN_low_legitimacy_12
			}
		}
		hidden_effect = { remove_ideas = MAN_low_legitimacy_12 }
	}
	if = {
		limit = { has_idea = MAN_low_legitimacy_22 }
		swap_ideas = {
			remove_idea = MAN_low_legitimacy_22
			add_idea = MAN_low_legitimacy_12
		}
	}
	if = {
		limit = { has_idea = MAN_low_legitimacy_32 }
		swap_ideas = {
			remove_idea = MAN_low_legitimacy_32
			add_idea = MAN_low_legitimacy_22
		}
	}
	if = {
		limit = { has_idea = MAN_low_legitimacy_42 }
		swap_ideas = {
			remove_idea = MAN_low_legitimacy_42
			add_idea = MAN_low_legitimacy_32
		}
	}
	if = {
		limit = { has_idea = MAN_low_legitimacy_52 }
		swap_ideas = {
			remove_idea = MAN_low_legitimacy_52
			add_idea = MAN_low_legitimacy_42
		}
	}
}
