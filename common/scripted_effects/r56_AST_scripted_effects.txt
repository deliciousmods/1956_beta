AST_set_robert_menzies = {
	if = {
		limit = {
			date > 1944.1.1
		}
		set_party_name = {
			ideology = democratic
			long_name = AST_liberal_party_long
			name = AST_liberal_party
		}
	}
	else = {
		set_party_name = {
			ideology = democratic
			long_name = AST_democratic_party_long
			name = AST_democratic_party
		}
	}
	
	create_country_leader = {
		name = "Robert Menzies"
		#desc = "POLITICS_ROBERT_MENZIES_DESC"
		picture = "r56_portrait_AST_Robert_Menzies.dds"
		expire = "1965.1.1"
		ideology = liberalism
		traits = {
			skilled_negotiator
			savvy_politician
		}
	}
}

AST_lower_depression = {
	custom_effect_tooltip = AST_lower_depression_tt
	if = {
		limit = {
			has_idea = AST_R56_great_depression_1
		}
		swap_ideas = {
			remove_idea = AST_R56_great_depression_1
			add_idea = AST_R56_great_depression_2
		}
	}
	else_if = {
		limit = {
			has_idea = AST_R56_great_depression_2
		}
		swap_ideas = {
			remove_idea = AST_R56_great_depression_2
			add_idea = AST_R56_great_depression_3
		}
	}
	else = {
		remove_ideas = AST_R56_great_depression_3
	}
}

AST_add_or_modify_economy_dm = {
	text = AST_add_or_modify_economy_tt
    if = {
        limit = { has_dynamic_modifier = AST_australian_economy_dm }
        # already exists ? do nothing
    }
	custom_effect_tooltip = AST_modify_australian_economy_tt
    else = {
        add_dynamic_modifier = { modifier = AST_australian_economy_dm }
    }
}

AST_upgrade_economy_to_recovery = {

    if = {
        limit = { has_dynamic_modifier = AST_depression_economy_dm }

        remove_dynamic_modifier = AST_depression_economy_dm
        add_dynamic_modifier    = { modifier = AST_recovery_economy_dm }
    }

    # If already recovery or recovered do nothing
}

AST_upgrade_economy_to_recovered = {

    if = {
        limit = { has_dynamic_modifier = AST_recovery_economy_dm }

        remove_dynamic_modifier = AST_recovery_economy_dm
        add_dynamic_modifier    = { modifier = AST_australian_economy_dm }
    }

	# If already recovered do nothing
}
