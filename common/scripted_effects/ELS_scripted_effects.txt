### Central American Offers ###
# Used by the 5 Central American offer events (el_salvador.1 - el_salvador.5) to send El Salvador a report
# for each country that answers. 

# Invited country, inside the accept option of an offer event
ELS_offer_accepted_response = {
	ELS = { country_event = { id = el_salvador.10 } }
}

# Invited country, inside the decline option of an offer event
ELS_offer_declined_response = {
	ELS = { country_event = { id = el_salvador.11 } }
}

### United States of Central America ###
# Called when the first country joins the federation (Draft A Federal Constitution, el_salvador.5
# or one of the integration decisions). Only does anything the first time it is called.
ELS_form_central_america = {
	if = {
		limit = { NOT = { has_country_flag = ELS_central_america_formed } }
		# Reconstitute the Federal Republic fires the generic formable news instead
		if = {
			limit = { NOT = { has_country_flag = ELS_formed_by_decision } }
			if = {
				limit = { has_government = communism }
				news_event = { id = el_salvador.67 hours = 6 }
			}
			else = {
				news_event = { id = el_salvador.66 hours = 6 }
			}
		}
		set_country_flag = ELS_central_america_formed
		set_cosmetic_tag = ELS_USCA
		set_country_flag = USC_UNIFIED_FLAG
		# Stops the generic United Central America formable from firing again
		set_global_flag = form_united_central_america_flag
	}
}

#  Same as ELS_form_central_america but shows a tooltip, for use in decisions
ELS_form_central_america_tooltip = {
	if = {
		limit = { NOT = { has_country_flag = ELS_central_america_formed } }
		custom_effect_tooltip = ELS_form_central_america_tt
		hidden_effect = { ELS_form_central_america = yes }
	}
}

### Economy Law ###
# Moves the economy law up one step, stopping at War Economy
ELS_increase_economy_law = {
	if = {
		limit = { has_idea = civilian_economy }
		add_ideas = low_economic_mobilisation
	}
	else_if = {
		limit = { has_idea = low_economic_mobilisation }
		add_ideas = partial_economic_mobilisation
	}
	else_if = {
		limit = { has_idea = partial_economic_mobilisation }
		add_ideas = war_economy
	}
}
