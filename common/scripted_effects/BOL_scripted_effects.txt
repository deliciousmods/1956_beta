BOL_toro_leader = {
	set_politics = { ruling_party = neutrality elections_allowed = no }
	hidden_effect = {
		if = {
			limit = { has_character = BOL_david_toro }
			# Old saves may lack a role which is present in a fresh game.
			if = {
				limit = { NOT = { has_country_flag = BOL_toro_leader_role_ready } }
				add_country_leader_role = {
					character = BOL_david_toro
					promote_leader = yes
					country_leader = { ideology = military_junta expire = "1970.1.1" traits = { army_connections } }
				}
				set_country_flag = BOL_toro_leader_role_ready
			}
			promote_character = { character = BOL_david_toro ideology = military_junta }
		}
		else = {
			# Native construction can repair an old save without late recruit_character.
			create_country_leader = {
				name = BOL_david_toro
				picture = "gfx/leaders/BOL/r56_portrait_BOL_David_Toro.dds"
				expire = "1970.1.1"
				ideology = military_junta
				traits = { army_connections }
			}
		}
	}
}

		limit = { NOT = { has_country_flag = BOL_literacy_reform } }

		remove_ideas = BOL_rural_illiteracy
		add_stability = 0.02
	}
}

		add_extra_state_shared_building_slots = 1
		if = { limit = { infrastructure < 5 } add_building_construction = { type = infrastructure level = 1 instant_build = yes } }
		else = { ROOT = { add_political_power = 15 } }
	}
}

		limit = { NOT = { has_country_flag = BOL_pacific_shipyards } any_owned_state = { is_controlled_by = ROOT is_coastal = yes } }
		random_owned_controlled_state = {
			limit = { is_coastal = yes }
			add_extra_state_shared_building_slots = 2
			add_building_construction = { type = dockyard level = 2 instant_build = yes }
		}
		set_country_flag = BOL_pacific_shipyards
		remove_ideas = BOL_landlocked_navy
		add_ideas = BOL_titicaca_flotilla
		navy_experience = 15
	}
}

BOL_deliver_fighters = {
	if = {
		limit = { has_dlc = "By Blood Alone" }
		if = {
			limit = { NOT = { has_country_flag = BOL_fighters_model } }
			set_technology = { iw_small_airframe = 1 engines_1 = 1 aa_lmg = 1 early_bombs = 1 }
			create_equipment_variant = {
				name = "FAB Pursuit Aircraft"
				type = small_plane_airframe_0
				modules = { fixed_main_weapon_slot = light_mg_2x engine_type_slot = engine_1_1x }
			}
			set_country_flag = BOL_fighters_model
		}
		add_equipment_to_stockpile = { type = small_plane_airframe_0 amount = 24 producer = BOL variant_name = "FAB Pursuit Aircraft" }
	}
	else = { add_equipment_to_stockpile = { type = fighter_equipment_0 amount = 24 producer = BOL } }
}
