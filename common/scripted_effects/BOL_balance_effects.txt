# One active military programme: later choices replace the mass-levy stages.
BOL_refresh_ayllu_programme = {
 if = {
  limit = { has_government = communism has_country_flag = BOL_indigenous_backers }
  remove_ideas = { BOL_ayllu_mass_levies BOL_ayllu_peoples_army BOL_ayllu_road_army BOL_ayllu_confederate_levies BOL_falange_assault_cadres BOL_falange_hispanist_cadres BOL_falange_professional_army BOL_falange_andean_columns BOL_falange_andean_army BOL_workers_transition BOL_workers_planned_republic BOL_workers_revolutionary_state }
  remove_country_leader_trait = BOL_professional_army_advocate
  remove_country_leader_trait = BOL_andean_mobilizer
  if = {
   limit = { OR = { has_country_flag = BOL_ayllu_roads_choice has_completed_focus = BOL_the_inca_road_network_restored } }
   add_ideas = BOL_ayllu_road_army
  }
  else_if = {
   limit = { OR = { has_country_flag = BOL_ayllu_confederation_choice has_completed_focus = BOL_the_red_inca_confederation } }
   add_ideas = BOL_ayllu_confederate_levies
  }
  else_if = {
   limit = { OR = { has_country_flag = BOL_ayllu_military_organized has_completed_focus = BOL_the_aymara_peoples_army } }
   add_ideas = BOL_ayllu_peoples_army
  }
  else = { add_ideas = BOL_ayllu_mass_levies }
  set_country_flag = BOL_ayllu_balance_v7
 }
}

# Credit, provisional control and workers' boards give way to mature mining institutions.
# Keep the historical COMIBOL and industrial complex, but not their obsolete precursors.
BOL_refresh_mining_programme = {
 if = {
  limit = { OR = { has_idea = BOL_mining_state_complex has_idea = BOL_comibol_idea } }
  remove_ideas = { BOL_mining_credit_system BOL_state_mining_control BOL_miner_worker_control BOL_mine_joint_boards }
 }
 else_if = {
  limit = { has_idea = BOL_state_mining_control }
  remove_ideas = BOL_mining_credit_system
 }
 if = {
  limit = { has_idea = BOL_economic_miracle }
  remove_ideas = BOL_bohan_plan
 }
 set_country_flag = BOL_mining_balance_v7
}

# Fully developed states must not swallow an infrastructure reward.
BOL_build_core_infrastructure_1 = {
 if = {
  limit = { any_owned_state = { is_core_of = ROOT is_fully_controlled_by = ROOT infrastructure < 5 } }
  random_owned_controlled_state = {
   limit = { is_core_of = ROOT infrastructure < 5 }
   add_building_construction = { type = infrastructure level = 1 instant_build = yes }
  }
 }
 else = { add_political_power = 15 }
}

BOL_build_core_infrastructure_2 = {
 if = {
  limit = { any_owned_state = { is_core_of = ROOT is_fully_controlled_by = ROOT infrastructure < 4 } }
  random_owned_controlled_state = {
   limit = { is_core_of = ROOT infrastructure < 4 }
   add_building_construction = { type = infrastructure level = 2 instant_build = yes }
  }
 }
 else_if = {
  limit = { any_owned_state = { is_core_of = ROOT is_fully_controlled_by = ROOT infrastructure < 5 } }
  BOL_build_core_infrastructure_1 = yes
  add_political_power = 15
 }
 else = { add_political_power = 30 }
}
