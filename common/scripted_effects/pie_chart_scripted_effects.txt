# this creates a pie chart of popularities - can be easily adapted to show anything that adds up to 100%

create_pie_chart = {
	clear_array = pie_chart
	resize_array = {
		array = pie_chart
		value = 1 # 1 is value for neutrality frame, so we don't have to make a loop for it
		size = 100 # 100 pieces, each percent is one piece
	}

	set_temp_variable = { nraf_pop = NRAF_seats_f }
	divide_temp_variable = { nraf_pop = GER_total_seats }
	multiply_temp_variable = { nraf_pop = 100 }
	round_temp_variable = nraf_pop
	set_variable = { ROOT.last_idx = nraf_pop }

	set_temp_variable = { dnvp_pop = DNVP_seats_f }
	divide_temp_variable = { dnvp_pop = GER_total_seats }
	multiply_temp_variable = { dnvp_pop = 100 }
	round_temp_variable = dnvp_pop
	add_to_temp_variable = { dnvp_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = dnvp_pop
		set_variable = { pie_chart^v = 2 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { dvu_pop = DVU_seats_f }
	divide_temp_variable = { dvu_pop = GER_total_seats }
	multiply_temp_variable = { dvu_pop = 100 }
	round_temp_variable = dvu_pop
	add_to_temp_variable = { dvu_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = dvp_pop
		set_variable = { pie_chart^v = 3 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { dvp_pop = DVP_seats_f }
	divide_temp_variable = { dvp_pop = GER_total_seats }
	multiply_temp_variable = { dvp_pop = 100 }
	round_temp_variable = dvp_pop
	add_to_temp_variable = { dvp_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = dvp_pop
		set_variable = { pie_chart^v = 4 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { zentrum_pop = Zentrum_seats_f }
	divide_temp_variable = { zentrum_pop = GER_total_seats }
	multiply_temp_variable = { zentrum_pop = 100 }
	round_temp_variable = zentrum_pop
	add_to_temp_variable = { zentrum_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = zentrum_pop
		set_variable = { pie_chart^v = 5 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { dstp_pop = DSTP_seats_f }
	divide_temp_variable = { dstp_pop = GER_total_seats }
	multiply_temp_variable = { dstp_pop = 100 }
	round_temp_variable = dstp_pop
	add_to_temp_variable = { dstp_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = dstp_pop
		set_variable = { pie_chart^v = 6 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { spd_pop = SPD_seats_f }
	divide_temp_variable = { spd_pop = GER_total_seats }
	multiply_temp_variable = { spd_pop = 100 }
	round_temp_variable = spd_pop
	add_to_temp_variable = { spd_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = spd_pop
		set_variable = { pie_chart^v = 7 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { sapd_pop = SAPD_seats_f }
	divide_temp_variable = { sapd_pop = GER_total_seats }
	multiply_temp_variable = { sapd_pop = 100 }
	round_temp_variable = sapd_pop
	add_to_temp_variable = { sapd_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = sapd_pop
		set_variable = { pie_chart^v = 8 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { kpd_pop = KPD_seats_f }
	divide_temp_variable = { kpd_pop = GER_total_seats }
	multiply_temp_variable = { kpd_pop = 100 }
	round_temp_variable = kpd_pop
	add_to_temp_variable = { kpd_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = kpd_pop
		set_variable = { pie_chart^v = 9 }
	}

	clear_variable = ROOT.last_idx
	add_to_variable = { pie_chart_dirty = 1 }
}
