# Only A Second Chance starts a civil war.
BOL_prepare_civil_war_militia = {
	if = {
		limit = { NOT = { has_template = "Milicias de Emergencia" } }
		division_template = {
			name = "Milicias de Emergencia"
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
			}
		}
	}
}

BOL_spawn_civil_war_militia = {
	random_owned_controlled_state = {
		create_unit = {
			division = "name = \"Milicia Boliviana\" division_template = \"Milicias de Emergencia\" start_experience_factor = 0.15 start_equipment_factor = 0.85"
			owner = PREV
		}
	}
}

BOL_start_second_civil_war = {
	if = {
		limit = { NOT = { has_country_flag = BOL_second_civil_war_started } has_civil_war = no }
		BOL_prepare_civil_war_militia = yes
		remove_ideas = BOL_unarmed_army
		start_civil_war = {
			ideology = democratic
			size = 0.3
			army_ratio = 0.3
			air_ratio = 0.25
			capital = 487
			states = { 487 }
			keep_all_characters = yes
			set_country_flag = BOL_second_loyalists
			set_country_flag = BOL_r56_political_timeline
			set_country_flag = BOL_second_civil_war_started
			clr_country_flag = BOL_nationalist_backers
			clr_country_flag = BOL_indigenous_backers
			drop_cosmetic_tag = yes
			add_manpower = 1500
			add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 300 producer = BOL }
			BOL_spawn_civil_war_militia = yes
			BOL_spawn_civil_war_militia = yes
		}
		set_country_flag = BOL_second_civil_war_started
		set_country_flag = BOL_second_revolutionaries
		add_manpower = 4000
		add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 1000 producer = BOL }
		if = {
			limit = { has_country_flag = BOL_indigenous_backers }
			BOL_raise_ayllu_guard = yes
		}
		else = {
			BOL_spawn_civil_war_militia = yes
			BOL_spawn_civil_war_militia = yes
		}
		add_timed_idea = { idea = BOL_revolutionary_momentum days = 120 }
		activate_mission = BOL_win_the_second_civil_war
	}
}
