BOL_establish_ayllu_command = {
 BOL_refresh_ayllu_programme = yes
 set_country_flag = BOL_ayllu_command_established
 if = { limit = { has_country_flag = BOL_second_civil_war_started } BOL_raise_ayllu_guard = yes }
}
BOL_raise_ayllu_guard = {
 if = {
  limit = { NOT = { has_country_flag = BOL_ayllu_guard_raised } any_owned_state = { is_fully_controlled_by = ROOT } }
  set_country_flag = BOL_ayllu_guard_raised
  if = {
   limit = { NOT = { has_template = "Guardia del Ayllu" } }
   division_template = {
    name = "Guardia del Ayllu"
    regiments = {
     infantry = { x = 0 y = 0 } infantry = { x = 0 y = 1 }
     infantry = { x = 1 y = 0 } infantry = { x = 1 y = 1 }
    }
   }
  }
  random_owned_controlled_state = {
   create_unit = {
    division = "name = \"Guardia de Qallapa\" division_template = \"Guardia del Ayllu\" start_experience_factor = 0.10 start_equipment_factor = 0.75"
    owner = PREV
   }
  }
  random_owned_controlled_state = {
   create_unit = {
    division = "name = \"Guardia de Omasuyos\" division_template = \"Guardia del Ayllu\" start_experience_factor = 0.10 start_equipment_factor = 0.75"
    owner = PREV
   }
  }
  random_owned_controlled_state = {
   create_unit = {
    division = "name = \"Guardia de Tiwanaku\" division_template = \"Guardia del Ayllu\" start_experience_factor = 0.10 start_equipment_factor = 0.75"
    owner = PREV
   }
  }
 }
}

BOL_campaign_PRU = {
 every_state = { limit = { is_core_of = PRU NOT = { is_core_of = ROOT } } add_claim_by = ROOT }
 if = {
  limit = { country_exists = PRU is_subject = no NOT = { has_war_with = PRU } NOT = { is_in_faction_with = PRU } PRU = { NOT = { is_subject_of = ROOT } } }
  create_wargoal = { type = annex_everything target = PRU }
 }
}
