r56_scripted_effect_PRU_state_development_level_up = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = r56_moderate_economic_inequality_modifier }
		}
		remove_dynamic_modifier = {
			modifier = r56_moderate_economic_inequality_modifier
		}
	}
	if = {
		limit = {
			has_dynamic_modifier = { modifier = r56_severe_economic_inequality_modifier }
		}
		remove_dynamic_modifier = {
			modifier = r56_severe_economic_inequality_modifier
		}
		add_dynamic_modifier = {
			modifier = r56_moderate_economic_inequality_modifier
		}
	}
	if = {
		limit = {
			has_dynamic_modifier = { modifier = r56_critical_economic_inequality_modifier }
		}
		remove_dynamic_modifier = {
			modifier = r56_critical_economic_inequality_modifier
		}
		add_dynamic_modifier = {
			modifier = r56_severe_economic_inequality_modifier
		}
	}
}

r56_scripted_effect_PRU_state_development_level_down = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = r56_severe_economic_inequality_modifier }
		}
		remove_dynamic_modifier = {
			modifier = r56_severe_economic_inequality_modifier
		}
		add_dynamic_modifier = {
			modifier = r56_critical_economic_inequality_modifier
		}
	}
	if = {
		limit = {
			has_dynamic_modifier = { modifier = r56_moderate_economic_inequality_modifier }
		}
		remove_dynamic_modifier = {
			modifier = r56_moderate_economic_inequality_modifier
		}
		add_dynamic_modifier = {
			modifier = r56_severe_economic_inequality_modifier
		}
	}
	if = {
		limit = {
            NOT = {
			    has_dynamic_modifier = { modifier = r56_moderate_economic_inequality_modifier }
			    has_dynamic_modifier = { modifier = r56_severe_economic_inequality_modifier }
			    has_dynamic_modifier = { modifier = r56_critical_economic_inequality_modifier }
            }
		}
		add_dynamic_modifier = {
			modifier = r56_moderate_economic_inequality_modifier
		}
	}
}
