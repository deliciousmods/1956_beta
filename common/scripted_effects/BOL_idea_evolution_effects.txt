# Evolving programmes. One idea per family; late completion cannot downgrade it.
# Each adoption uses swap_ideas where a predecessor exists, matching the Peru tree.

# mining
BOL_adopt_mining_credit_system = {
 if = {
  limit = { NOT = { OR = { has_idea = BOL_state_mining_control has_idea = BOL_miner_worker_control has_idea = BOL_mine_joint_boards has_idea = BOL_comibol_idea has_idea = BOL_mining_state_complex } } }
  if = {
   limit = { NOT = { has_idea = BOL_mining_credit_system } }
   add_ideas = BOL_mining_credit_system
  }
 }
}
BOL_adopt_state_mining_control = {
 if = {
  limit = { NOT = { OR = { has_idea = BOL_miner_worker_control has_idea = BOL_mine_joint_boards has_idea = BOL_comibol_idea has_idea = BOL_mining_state_complex } } }
  if = {
   limit = { NOT = { has_idea = BOL_state_mining_control } }
   if = {
    limit = { has_idea = BOL_mining_credit_system }
    swap_ideas = { remove_idea = BOL_mining_credit_system add_idea = BOL_state_mining_control }
   }
   else = { add_ideas = BOL_state_mining_control }
  }
  hidden_effect = { remove_ideas = { BOL_mining_credit_system } }
 }
}
