r56_SAM_increase_japanese_influence = {
	if = {
		limit = { NOT = { has_idea = r56_SAM_japanese_influence_i } }
		add_ideas = r56_SAM_japanese_influence_i
	}

	if = {
		limit = { has_idea = r56_SAM_japanese_influence_i }
		swap_ideas = {
			remove_idea = r56_SAM_japanese_influence_i
			add_idea = r56_SAM_japanese_influence_ii
		}
	}

	if = {
		limit = { has_idea = r56_SAM_japanese_influence_ii }
		swap_ideas = {
			remove_idea = r56_SAM_japanese_influence_ii
			add_idea = r56_SAM_japanese_influence_iii
		}
	}

	if = {
		limit = { has_idea = r56_SAM_japanese_influence_iii }
		swap_ideas = {
			remove_idea = r56_SAM_japanese_influence_iii
			add_idea = r56_SAM_japanese_influence_iv
		}
	}

	if = {
		limit = { has_idea = r56_SAM_japanese_influence_iv }
		swap_ideas = {
			remove_idea = r56_SAM_japanese_influence_iv
			add_idea = r56_SAM_japanese_influence_v
		}
	}
}

r56_SAM_decrease_japanese_influence = {
