# Unlocked by Central American Revolutionary Front. All four neighbors are single-state countries, so a civil war isn't possible (sadly)

ELS_revolutionary_front = {

	ELS_stage_communist_coup = {

		icon = generic_ignite_civil_war

		allowed = {
			original_tag = ELS
		}

		targets = { GUA HON NIC COS PAN }

		target_trigger = {
			FROM = { exists = yes }
			NOT = { has_war_with = FROM }
			OR = {
				NOT = { FROM = { original_tag = PAN } }
				has_completed_focus = ELS_revolution_in_panama
			}
		}

		available = {
			FROM = {
				NOT = { has_government = communism }
				is_subject = no # the coup puppets them, so don't take another country's subject
				custom_trigger_tooltip = {
					tooltip = ELS_coup_conditions_tt
					OR = {
						communism > 0.5
						AND = {
							communism > 0.3
							has_stability < 0.2
						}
					}
				}
			}
		}

		cost = 50

		fire_only_once = no

		ai_will_do = {
			factor = 1
		}

		complete_effect = {
			# Panama is only targetable after Revolution in Panama, when the union already exists, so it joins directly
			if = {
				limit = { FROM = { original_tag = PAN } }
				annex_country = { target = FROM transfer_troops = yes }
				every_owned_state = {
					limit = {
						is_core_of = PAN
						NOT = { is_core_of = ROOT }
					}
					add_core_of = ROOT
				}
			}
			else = {
				FROM = {
					set_politics = {
						ruling_party = communism
						elections_allowed = no
					}
					add_popularity = {
						ideology = communism
						popularity = 0.1
					}
					country_event = { id = el_salvador.23 }
				}
				# The new government becomes our puppet, subjects join their overlord's faction automatically
				puppet = FROM
			}
		}
	}
}

# Replaces form_united_central_america for El Salvador, so One Central America can be taken afterwards
ELS_reconstitute_the_federal_republic_category = {

	ELS_reconstitute_the_federal_republic = {

		icon = generic_form_nation

		allowed = {
			original_tag = ELS
		}

		visible = {
			NOT = { has_global_flag = form_united_central_america_flag }
			NOT = { has_country_flag = ELS_central_america_formed }
		}

		available = {
			is_subject = no
			controls_state = 313
			controls_state = 314
			controls_state = 312
			controls_state = 317
			controls_state = 316
		}

		ai_will_do = {
			factor = 200
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}

		complete_effect = {
			hidden_effect = { set_country_flag = ELS_formed_by_decision }
			ELS_form_central_america_tooltip = yes
			every_state = {
				limit = {
					NOT = { is_core_of = ROOT }
					OR = {
						state = 313
						state = 314
						state = 312
						state = 317
						state = 316
					}
				}
				add_core_of = ROOT
			}
			hidden_effect = {
				if = {
					limit = { has_dlc = "La Resistance" }
					every_country = {
						every_operative = {
							limit = {
								OR = {
									has_nationality = GUA
									has_nationality = HON
									has_nationality = NIC
									has_nationality = COS
								}
								NOT = { has_nationality = ROOT }
							}
							add_nationality = ROOT
						}
					}
				}
				news_event = { id = wtt_news.59 hours = 6 }
			}
		}
	}
}
