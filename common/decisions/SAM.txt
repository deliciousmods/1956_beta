SAM_goodwill_mission_negotiations_tracker_category = {
	SAM_mission_duration = {
		icon = GFX_decision_generic_speech
		days_mission_timeout = 40
		priority = 5

		available = {
			hidden_trigger = { 
				always = no
			}
		}

		visible = {
			has_country_flag = SAM_goodwill_mission_arrives
			OR = {
				original_tag = SAM
				original_tag = NZL
			}
			SAM = {
				exists = yes
				is_puppet_of = NZL
			}
			NZL = {
				exists = yes
			}
		}

		activation = {
			has_country_flag = SAM_goodwill_mission_arrives
		}

		fire_only_once = yes
		cancel_if_not_visible = no
		is_good = no
		
		timeout_effect = {
			clr_country_flag = SAM_goodwill_mission_arrives
			set_country_flag = SAM_goodwill_mission_concluded
			if = { # Ended in Samoan favor aka samoan protectorateship
				limit = {
					check_variable = {
						var = NZL.SAM_new_zealand_negotiations_influence 
						value = SAM.SAM_mau_negotiations_influence 
						compare = less_than
					}
					NOT = {
						OR = {
							has_country_flag = SAM_goodwill_mission_concludes_infavour_of_new_zealand
							has_country_flag = SAM_goodwill_mission_concludes_with_a_compromise
						}
					}
				}
				set_country_flag = SAM_goodwill_mission_concludes_infavour_of_mau
				SAM = {
					country_event = { id = goodwill_outcome.3  } #MAU NUMBA 1!!!!!!!!!
				}
				NZL = {
					country_event = { id = goodwill_outcome.4  } #No more malaria!
				}
			}
			else_if = { # In favour of NZL, aka independence
				limit = {
					NOT = {
						OR = {
							has_country_flag = SAM_goodwill_mission_concludes_infavour_of_mau
							has_country_flag = SAM_goodwill_mission_concludes_with_a_compromise
						}
					}
					check_variable = {
						var = NZL.SAM_new_zealand_negotiations_influence 
						value = SAM.SAM_mau_negotiations_influence 
						compare = greater_than
					}
				}
				set_country_flag = SAM_goodwill_mission_concludes_infavour_of_new_zealand
				SAM = {
					country_event = { id = goodwill_outcome.6  } # ratio
				}
				NZL = {
					country_event = { id = goodwill_outcome.5  } # it was at this moment he knew he fucked up
				}
			}
			else_if = { #Compromise
				limit = {
					NOT = {
						OR = {
							has_country_flag = SAM_goodwill_mission_concludes_infavour_of_new_zealand
							has_country_flag = SAM_goodwill_mission_concludes_infavour_of_mau
						}
					}
					check_variable = {
						var = NZL.SAM_new_zealand_negotiations_influence 
						value = SAM.SAM_mau_negotiations_influence 
						compare = equals
					}
				}
				set_country_flag = SAM_goodwill_mission_compromise #Aka samoa goes colonial
				SAM = {
					country_event = { id = goodwill_outcome.1  } # piss
				}
				NZL = {
					country_event = { id = goodwill_outcome.2  } # piss
				}
			}
		}

		complete_effect = {
			country_event = { id = egypt.17 hours = 12 } #Let the winner gain a cookie!
		}
	}
