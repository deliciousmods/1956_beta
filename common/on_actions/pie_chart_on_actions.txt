# This is just an example of how to update the pie chart, it does not need to be an on_action
on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				set_variable = { NRAF_seats = 25 }
				set_variable = { DNVP_seats = 40 }
				set_variable = { DVP_seats = 30}
				set_variable = { DSTP_seats = 30 }
				set_variable = { Zentrum_seats = 60 }
				set_variable = { SAPD_seats = 30 }
				set_variable = { SPD_seats = 70 }
				set_variable = { KPD_seats = 30 }
				create_pie_chart = yes
			}
		}
	}
}
