	on_capitulation = {
		effect = {
			if = {
				limit = {
					AND = {
						FROM = {
							OR = {
								is_in_faction = no
								is_faction_leader = yes
							}
						}
						ROOT = { has_idea = KMT_chinese_major } 	# if China or a successor capitulated to a warlord
						OR = {
							FROM = { original_tag = PRC }
							FROM = { original_tag = MAN }
							FROM = { original_tag = MEN }
							FROM = { original_tag = SHX }
							FROM = { original_tag = XSM }
							FROM = { original_tag = SIK }
							FROM = { original_tag = TNG }
							FROM = { original_tag = YUN }
							FROM = { original_tag = GXC }
						}
					}
				}
				ROOT = { remove_ideas =KMT_chinese_major set_major = no } 		# loser loses idea (capitulation does not mean full annexation)
				FROM = { add_ideas = KMT_chinese_major     set_major = yes } 	# successor gets idea
			}
		}
