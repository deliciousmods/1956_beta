	on_daily_ROM = {
		effect = {
			for_each_scope_loop = {
				array = ROM.unavailable_diplomacy_partners
				if = {
					limit = {
						exists = yes
						NOT = {
							is_in_array = {	#trying to fix repeat firing
								array = global.rom_diplomacy_partners
								value = THIS.id
							}
						}
					}
					add_to_array = {
						array = global.rom_diplomacy_partners
						value = THIS.id
					}
					remove_from_array = {
						array = ROM.unavailable_diplomacy_partners
						value = THIS.id
					}
				}
			}
			for_each_scope_loop = {
				array = ROM.blocked_diplo_countries
				#are they an ally while being on the block list? remove them
				if = {
					limit = {
						is_ally_with = ROM
					}
					remove_from_array = {
						array = ROM.blocked_diplo_countries
						value = THIS.id
					}
				}
			}
			for_each_scope_loop = {
				array = global.rom_diplomacy_partners
				#if they stopped existing? remove them
				if = {
					limit = {
						exists = no
						NOT = {
							is_in_array = {	#trying to fix repeat firing
								array = ROM.unavailable_diplomacy_partners
								value = THIS.id
							}
						}
					}
					set_variable = { ROM_relation_var = 1 }
					if = {
						limit = {
							ROM_is_improving_relations = yes
						}
						ROM_diplo_stop_improving_tier = yes
					}
					remove_from_array = {
						array = global.rom_diplomacy_partners
						value = THIS.id
					}
					add_to_array = {
						array = ROM.unavailable_diplomacy_partners
						value = THIS.id
					}
				}
				#for optimization, check if anyone changed tier and only then run the bonus recalculations
				if = {
					limit = {
						check_variable = { ROM_relation_var > 4.5 }
					}
					set_temp_variable = { ROM_old_tier = 5 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 3.5 }
					}
					set_temp_variable = { ROM_old_tier = 4 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 2.5 }
					}
					set_temp_variable = { ROM_old_tier = 3 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 1.5 }
					}
					set_temp_variable = { ROM_old_tier = 2 }
				}
				else = {
					set_temp_variable = { ROM_old_tier = 1 }
				}
				#decay
				set_variable = { decay_multiplier = opinion@ROM }	#range: -100 - +100
				multiply_variable = { decay_multiplier = -0.005 }	#range: +0.5 - -0.5
				add_to_variable = { decay_multiplier = 1 }			#range: +1.5 - +0.5
				if = {
					limit = {
						check_variable = {
							ROM_relation_var > 3.5
						}
					}
					#subtract_from_variable = { ROM_relation_var = 0.0066 }
					set_temp_variable = { relation_loss = 0.0066 }
					multiply_temp_variable = { relation_loss = decay_multiplier }
					subtract_from_variable = { ROM_relation_var = relation_loss }
				}
				else_if = {
					limit = {
						check_variable = {
							ROM_relation_var > 1.003
						}
					}
					#subtract_from_variable = { ROM_relation_var = 0.0033 }
					set_temp_variable = { relation_loss = 0.0033 }
					multiply_temp_variable = { relation_loss = decay_multiplier }
					subtract_from_variable = { ROM_relation_var = relation_loss }
				}
				else_if = {
					limit = {
						check_variable = {
							ROM_relation_var > 1
						}
					}
					set_variable = {
						ROM_relation_var = 1
					}
				}
				#improving relations
				if = {
					limit = {
						ROM_is_improving_relations = yes
					}
					if = {
						limit = {
							ROOT = {
								check_variable = {	#political power is decreasing daily
									political_power_daily < 0
								}
								set_temp_variable = {
									neg_political_power_daily = 0
								}
								subtract_from_temp_variable = {
									neg_political_power_daily = political_power_daily
								}
								NOT = {
									check_variable = {	#our daily decrease must be smaller than our current reserve
										neg_political_power_daily < political_power
									}
								}
							}
						}
						ROM_diplo_stop_improving_tier = yes
					}
					else = {
						add_to_variable = { ROM_relation_var = 0.02 }
						if = {
							limit = {
								check_variable = {
									ROM_relation_var > 5.5
								}
							}
							set_variable = { ROM_relation_var = 5.5 }	#prevent building to huge values
							#ROM_diplo_stop_improving_tier = yes
							if = {
								limit = {
									NOT = {
										has_country_flag = ROM_maintaining_relations_high
									}
								}
								clr_country_flag = ROM_improving_relations
								set_country_flag = ROM_maintaining_relations_high
								ROOT = {
									subtract_from_variable = {
										ROM_international_relations_political_power_cost = ROM_one_country_relations_boosting_cost
									}
									add_to_variable = {
										ROM_international_relations_political_power_cost = ROM_one_country_relations_maintain_cost
									}
								}
							}
						}
						else_if = {	#if focuses/events lowered the relation, go back to normal boosting
							limit = {
								has_country_flag = ROM_maintaining_relations_high
							}
							set_country_flag = ROM_improving_relations
							clr_country_flag = ROM_maintaining_relations_high
							ROOT = {
								add_to_variable = {
									ROM_international_relations_political_power_cost = ROM_one_country_relations_boosting_cost
								}
								subtract_from_variable = {
									ROM_international_relations_political_power_cost = ROM_one_country_relations_maintain_cost
								}
							}
						}
					}
				}
				#see if the dynamic modifier needs to be refreshed
				if = {
					limit = {
						check_variable = { ROM_relation_var > 4.5 }
					}
					set_temp_variable = { ROM_new_tier = 5 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 3.5 }
					}
					set_temp_variable = { ROM_new_tier = 4 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 2.5 }
					}
					set_temp_variable = { ROM_new_tier = 3 }
				}
				else_if = {
					limit = {
						check_variable = { ROM_relation_var > 1.5 }
					}
					set_temp_variable = { ROM_new_tier = 2 }
				}
				else = {
					set_temp_variable = { ROM_new_tier = 1 }
				}
				if = {
					limit = {
						check_variable = {
							var = ROM_new_tier
							value = ROM_old_tier
							compare = not_equals
						}
					}
					set_temp_variable = {
						ROM_modifier_must_refresh = 1
					}
					if = {
						limit = {
							OR = {
								has_idea = ROM_relation_with_romania_2
								has_idea = ROM_relation_with_romania_3
								has_idea = ROM_relation_with_romania_4
								has_idea = ROM_relation_with_romania_5
							}
						}
						remove_ideas = {
							ROM_relation_with_romania_2
							ROM_relation_with_romania_3
							ROM_relation_with_romania_4
							ROM_relation_with_romania_5
						}
						if = {
							limit = {
								check_variable = { ROM_relation_var > 4.5 }
							}
							add_ideas = ROM_relation_with_romania_5
						}
						else_if = {
							limit = {
								check_variable = { ROM_relation_var > 3.5 }
							}
							add_ideas = ROM_relation_with_romania_4
						}
						else_if = {
							limit = {
								check_variable = { ROM_relation_var > 2.5 }
							}
							add_ideas = ROM_relation_with_romania_3
						}
						else_if = {
							limit = {
								check_variable = { ROM_relation_var > 1.5 }
							}
							add_ideas = ROM_relation_with_romania_2
						}
					}
				}
			}
			if = {
				limit = {
					check_variable = {
						ROM_modifier_must_refresh > 0
					}
				}
				ROM_update_foreign_relations_dynamic_modifier = yes
			}
		}
	}

										original_tag = ITA
										original_tag = GRE
									}
								}
							}
						}
						if = {
							limit = {
								amount_taken_ideas = {
									amount > 2
									slots = {
										political_advisor
									}
								}
								OR = {	#AI can't time this well, so spend at most 0.5pp daily
									check_variable = {
										ROM_current_improving_countries < 2
									}
									AND = {
										check_variable = {
											ROM_current_improving_countries < 3
										}
										OR = {
											has_idea_with_trait = ROM_the_white_prince
											has_idea_with_trait = ROM_camarilla_council_member_2
											has_idea_with_trait = ROM_pragmatic_diplomat
										}
									}
								}
								NOT = {
									ROM_is_improving_relations = yes
								}
							}
							ROM_diplo_start_improving_tier = yes
						}
					}
					else_if = {
						limit = {
							ROM_is_improving_relations = yes
						}
						ROM_diplo_stop_improving_tier = yes
					}
				}
			}
		}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					FROM = {
						original_tag = ROM
						OR = {
							has_completed_focus = ROM_maintain_regional_stability
							has_completed_focus = ROM_dominance_over_the_balkans
						}
					}
				}
				every_core_state = {
					limit = {
						is_owned_by = ROOT
					}
					add_to_temp_variable = {
						ROOT.casualties_max = state_population_k
					}
				}
				divide_temp_variable = {
					casualties_max = 100
				}
				set_temp_variable = {
					casualties_max_low = casualties_max
				}
				divide_temp_variable = {
					casualties_max_low = 2
				}
				#log = "Max casualties in k: [?casualties_max]"
				if = {
					limit = {
						FROM = {
							has_completed_focus = ROM_dominance_over_the_balkans
						}
					}
					if = {
						limit = {
							casualties_inflicted_by = {
								opponent = FROM
								thousands < casualties_max
							}
						}
						FROM = {
							every_controlled_state = {
								limit = {
									is_core_of = ROOT
								}
								add_compliance = 15
							}
						}
					}
					if = {
						limit = {
							casualties_inflicted_by = {
								opponent = FROM
								thousands < casualties_max_low
							}
						}
						FROM = {
							add_to_variable = {
								ROM_countries_defeated_low_casualties = 1
							}
							every_controlled_state = {
								limit = {
									is_core_of = ROOT
								}
